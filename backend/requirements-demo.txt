# Minimal, fully installable dependency set for running ARCADIA-9 locally.
#
# `requirements.txt` in this directory is the original cloud image lock file: it
# pins a private wheel hosted on an internal domain
# (customer-assets.emergentagent.com), so `pip install -r requirements.txt`
# cannot succeed outside that environment. Use this file instead for local work.
#
#     pip install -r requirements-demo.txt
#
# Nothing here is paid and no API key is required: the simulation engine is
# fully local and the default persistence is a JSON file.

uvicorn[standard]>=0.25
pydantic>=2.6
python-dotenv>=1.0
requests>=2.31

# Testing
pytest>=8.0

# Optional extras. Uncomment the ones you want; the game runs without them.
# The backend detects each one at start-up and falls back to a stdlib
# implementation when it is absent.
#   motor + pymongo -> MongoDB instead of the local JSON save file
#   bcrypt          -> bcrypt password hashing instead of the PBKDF2 fallback
#   PyJWT           -> PyJWT tokens instead of the stdlib HMAC implementation
#   stripe          -> real Stripe checkout (payments are switched off by default)

# pymongo>=4.6
